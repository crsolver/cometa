// Scalar easing curves inspired by Odin core:math/ease and AHEasing.
// Public entry points clamp progress, preserve overshoot and return exact endpoints.
// NaN propagates; infinities map to their respective endpoints.
func _hgcurva_lineal(p float64) float64 {
 if p <= 0 { return 0 }; if p >= 1 { return 1 }; return p
}

func _hgcurva_cuadratica_entrada(p float64) float64 {
 if p <= 0 { return 0 }; if p >= 1 { return 1 }; return _hgcurvaBase_cuadratica(p)
}

func _hgcurva_cuadratica_salida(p float64) float64 {
 if p <= 0 { return 0 }; if p >= 1 { return 1 }; return 1 - _hgcurvaBase_cuadratica(1-p)
}

func _hgcurva_cuadratica_entrada_salida(p float64) float64 {
 if p <= 0 { return 0 }; if p >= 1 { return 1 }; if p < 0.5 { return 0.5 * _hgcurvaBase_cuadratica(2*p) }; return 1 - 0.5 * _hgcurvaBase_cuadratica(2-2*p)
}

// Entrance curves; exit and entrance-exit use reflection and half-interval scaling.
func _hgcurvaBase_cuadratica(p float64) float64 { return p*p }
func _hgcurvaBase_cubica(p float64) float64 { return p*p*p }
func _hgcurvaBase_cuartica(p float64) float64 { return p*p*p*p }
func _hgcurvaBase_quintica(p float64) float64 { return p*p*p*p*p }
func _hgcurvaBase_senoidal(p float64) float64 { return 1-math.Cos(p*math.Pi/2) }
func _hgcurvaBase_circular(p float64) float64 { return 1-math.Sqrt(1-p*p) }
func _hgcurvaBase_exponencial(p float64) float64 { return math.Exp2(10*(p-1)) }
func _hgcurvaBase_elastica(p float64) float64 { return math.Sin(13*math.Pi*p/2)*math.Exp2(10*(p-1)) }
func _hgcurvaBase_retroceso(p float64) float64 { return p*p*p-p*math.Sin(p*math.Pi) }
func _hgcurvaBase_rebote(p float64) float64 { return 1-_hgcurvaReboteSalida(1-p) }
func _hgcurvaReboteSalida(p float64) float64 {
 switch {
 case p < 4.0/11: return 121*p*p/16
 case p < 8.0/11: return 363.0/40*p*p-99.0/10*p+17.0/5
 case p < 9.0/10: return 4356.0/361*p*p-35442.0/1805*p+16061.0/1805
 default: return 54.0/5*p*p-513.0/25*p+268.0/25
 }
}
