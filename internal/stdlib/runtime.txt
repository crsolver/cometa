// Transform local geometry before applying the active camera. Image origins use
// source pixels; rectangle/destination origins use already-sized local pixels.
func _hgtransform(p,o,s _hgVec2,r float64,destinationOrigin bool) ebiten.GeoM {
 var m ebiten.GeoM
 if !destinationOrigin {m.Translate(-o.X,-o.Y)}
 m.Scale(s.X,s.Y)

 m.Rotate(r);m.Translate(p.X,p.Y);m.Concat(_hgcameraMatrix())
 return m
}

func _hgejecutar(instancia _hgJuego,w,h int64,title string,scale float64,resize,full bool,tps int64,pixelado,retro bool) struct {tag int; payload2 string} {
 fail:=func(err error) struct{tag int;payload2 string}{return struct{tag int;payload2 string}{2,err.Error()}}
 config:=_hgConfigJuego{w,h,title,scale,resize,full,tps}
 if err:=_hgvalidarConfig(config);err!=nil{return fail(err)}
 if _hgrunStarted{return fail(fmt.Errorf("Pincel solo permite ejecutar un juego una vez por proceso"))}
 var game ebiten.Game = &_hgGame{instancia:instancia}
 if retro {
  shader,err:=ebiten.NewShader([]byte(_hgCRTSource))
  if err!=nil{return fail(fmt.Errorf("no se pudo inicializar el efecto CRT: %w",err))}
  defer shader.Dispose()
  game=&_hgCRTGame{Game:game,shader:shader}
 }
 _hgrunStarted=true
 _hgconfig=config
 ebiten.SetWindowSize(int(float64(w)*scale),int(float64(h)*scale))
 ebiten.SetWindowTitle(title)
 ebiten.SetFullscreen(full)
 if resize {ebiten.SetWindowResizingMode(ebiten.WindowResizingModeEnabled)}
 ebiten.SetTPS(int(tps))
 ebiten.SetScreenFilterEnabled(!(pixelado||retro))
 if _hgcapturaRuta!="" {ebiten.SetWindowVisible(false);ebiten.SetRunnableOnUnfocused(true)}
 if err:=ebiten.RunGame(game);err!=nil{return fail(err)}
 return struct{tag int;payload2 string}{1,""}
}

func (i *_hgImagen) gpu() *ebiten.Image {

 if i.dirty {i.image.WritePixels(_hgpremultiplicar(i.pixels.Pix));i.dirty=false}
 return i.image
}
func (i *_hgImagen) bounds() image.Rectangle {if i.pixels!=nil{return i.pixels.Rect};return i.image.Bounds()}
func _hgpremultiplicar(pix []byte) []byte {
 out:=make([]byte,len(pix))
 for i:=0;i<len(pix);i+=4 {a:=uint32(pix[i+3]);out[i]=uint8(uint32(pix[i])*a/255);out[i+1]=uint8(uint32(pix[i+1])*a/255);out[i+2]=uint8(uint32(pix[i+2])*a/255);out[i+3]=pix[i+3]}
 return out
}
func _hgnrgba(src image.Image) *image.NRGBA {
 b:=src.Bounds();out:=image.NewNRGBA(image.Rect(0,0,b.Dx(),b.Dy()))
 for y:=0;y<b.Dy();y++ {for x:=0;x<b.Dx();x++ {out.Set(x,y,src.At(b.Min.X+x,b.Min.Y+y))}}
 return out
}

var _hgcapturaEscala int64 = 1
var _hgcapturaHecha bool
func _hgcapturar(screen *ebiten.Image) {
 if _hgcapturaRuta=="" || _hgcapturaHecha || _hgtick<_hgcapturaCuadros {return}
 b:=screen.Bounds();pix:=make([]byte,4*b.Dx()*b.Dy());screen.ReadPixels(pix)
 shot:=&image.RGBA{Pix:pix,Stride:4*b.Dx(),Rect:image.Rect(0,0,b.Dx(),b.Dy())}
 if err:=_hgescribirPNG(shot,_hgcapturaRuta,_hgcapturaEscala);err!=nil{log.Fatalf("captura: %v",err)}
 _hgcapturaHecha=true
}
